function getBonus() {
  const ss = SpreadsheetApp.openById('12p71jgMErzZYO4toU2LVELDPfwklHCyDARclldYImCc');   // DBC Base
  const sh = ss.getSheetByName('限時推薦獎金');
  if (!sh || sh.getLastRow() < 2) return { status: 'ok', rows: [] };
  // A~I 欄：類型 / 地區 / 開始日期 / 結束日期 / 組數 / 人數下限 / 人數上限 / 獎金 / 狀態
  // J 欄「內部備註」不會傳到 Portal。getDisplayValues 讓日期維持 yyyy/mm/dd 文字。
  return { status: 'ok', rows: sh.getRange(2, 1, sh.getLastRow() - 1, 9).getDisplayValues() };
}
